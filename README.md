## استفاده

```php
use Tavaan\Engine\Config\Repository;
use Tavaan\Engine\Config\EnvRepository;
use Tavaan\Engine\Config\EnvLoader;
use Tavaan\Engine\Config\ConfigLoader;

// بارگذاری پیکربندی
$loader = new ConfigLoader('config/');
$items = $loader->load();
$repo = new Repository($items);

// Dot-notation get/set
$repo->get('app.name');                    // 'Tavaan'
$repo->get('db.host', '127.0.0.1');
$repo->set('app.maintenance', true);
$repo->has('cache.stores.redis');
$repo->forget('app.debug');
$repo->pull('app.maintenance', false);     // get + forget
$repo->push('http.middleware.global', MyMiddleware::class);
$repo->prepend('http.middleware.global', OtherMiddleware::class);

// Namespace
$repo->namespace('mail');                 // ['default' => ..., 'mailers' => [...]]

// Freeze (در production)
$repo->freeze();
// $repo->set('x', 'y');  // throws ConfigFrozenException

// JSON با mask secretها
$json = $repo->toJson(maskSecrets: true);

// Env
$envRepo = new EnvRepository();
$envRepo->string('TVN_APP_NAME', 'Tavaan');
$envRepo->bool('TVN_APP_DEBUG', false);
$envRepo->int('TVN_PORT', 8000);
$envRepo->list('TVN_TRUSTED_PROXIES', ',');
$envRepo->json('TVN_EXTRA_CONFIG');
$envRepo->required(['TVN_APP_KEY']);       // throws if missing

// .env Loader
$envLoader = (new EnvLoader())->load('.env');
$envLoader->get('APP_NAME');
```
